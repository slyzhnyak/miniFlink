Module Checkpoint_parallel

type epoch = int
type 'a msg =
  1. | Event of 'a Mf_event.t
  2. | Barrier of epoch
type worker_snapshot = {
  1. worker : int;
  2. epoch : epoch;
  3. state : bytes;
}
type checkpoint_store = {
  1. mutable committed : (epoch * worker_snapshot array) list;
  2. cs_mu : Mutex.t;
}
val make_store : unit -> checkpoint_store
val commit : checkpoint_store -> epoch -> worker_snapshot array -> unit
val latest_checkpoint : checkpoint_store -> (epoch * worker_snapshot array) option
val checkpoint_count : checkpoint_store -> int
type coordinator = {
  1. workers : int;
  2. store : checkpoint_store;
  3. pending : worker_snapshot option array;
  4. co_mu : Mutex.t;
  5. co_done : Condition.t;
}
val make_coordinator : workers:int -> store:checkpoint_store -> coordinator
val submit_snapshot : coordinator -> worker_snapshot -> unit
val wait_committed : coordinator -> epoch:epoch -> unit
val hash_key : string -> int -> int
val run_exactly_once : workers:int -> capacity:int -> checkpoint_every:int -> key_of:('a -> string) -> make_state:(unit -> State_backend_memory.t) -> process:(State_backend_memory.t -> 'a0 Mf_event.t -> 'b list) -> source:'a1 Mf_event.t Stream.t -> sink:('b0 -> unit) -> store:checkpoint_store -> unit -> unit

run_exactly_once: ~workers число воркеров ~capacity размер каждого канала ~checkpoint_every барьер каждые N событий (по dispatcher) ~key_of ключ шардирования ~make_state создать backend для воркера (вызывается N раз) ~process обработать событие, обновляя backend; вернуть выходы ~key_of_out (не нужен) — выходы идут в sink ~source вход ~sink выход ~store куда коммитить checkpoint-ы

process получает (backend, событие) и возвращает список выходов. Стейт живёт в backend — именно он снапшотится на barrier.

val restore_latest : checkpoint_store -> State_backend_memory.t array -> bool
